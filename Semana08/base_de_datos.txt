-- =========================================
-- TABLAS
-- =========================================

CREATE TABLE Dependencia(
    IdDependencia     INT IDENTITY(1,1) PRIMARY KEY,
    NombreDependencia VARCHAR(100)
);

CREATE TABLE Tipo_Equipo(
    IdTipoEquipo     INT IDENTITY(1,1) PRIMARY KEY,
    NombreTipoEquipo VARCHAR(50)
);

CREATE TABLE Categoria_Software(
    IdCategoria     INT IDENTITY(1,1) PRIMARY KEY,
    NombreCategoria VARCHAR(50)
);

CREATE TABLE Software(
    IdSoftware    INT IDENTITY(1,1) PRIMARY KEY,
    NombreSoftware VARCHAR(100),
    IdCategoria   INT,
    FOREIGN KEY (IdCategoria) REFERENCES Categoria_Software(IdCategoria)
);

CREATE TABLE Tipo_Parte(
    IdTipoParte    INT IDENTITY(1,1) PRIMARY KEY,
    NombreTipoParte VARCHAR(50)
);

CREATE TABLE Equipo(
    IdEquipo      INT IDENTITY(1,1) PRIMARY KEY,
    IdDependencia INT,
    Fecha         DATE,
    UsuarioEquipo VARCHAR(100),
    IdTipoEquipo  INT,
    MarcaModelo   VARCHAR(100),
    NroSerieCPU   VARCHAR(100),
    Procesador    VARCHAR(100),
    -- *** CAMBIO: Velocidad ahora es DECIMAL en GHz ***
    VelocidadGHz  DECIMAL(4,2),
    -- *** CAMBIO: RAM ahora es INT en GB ***
    RAM_GB        INT,
    TipoDiscoDuro VARCHAR(10),   -- SSD / HDD
    -- *** CAMBIO: Capacidad ahora es INT en GB ***
    CapacidadGB   INT,
    GPU           VARCHAR(50),
    -- MemoriaGPU puede ser 'Integrada' o número, se deja VARCHAR
    MemoriaGPU    VARCHAR(20),
    EstadoEquipo  VARCHAR(50),
    Observacion   VARCHAR(200),
    -- *** NUEVO: precio referencial para demos de división ***
    Precio        DECIMAL(10,2),

    FOREIGN KEY (IdDependencia) REFERENCES Dependencia(IdDependencia),
    FOREIGN KEY (IdTipoEquipo)  REFERENCES Tipo_Equipo(IdTipoEquipo)
);

CREATE TABLE Equipo_Software(
    IdEquipo   INT,
    IdSoftware INT,
    PRIMARY KEY(IdEquipo, IdSoftware),
    FOREIGN KEY (IdEquipo)   REFERENCES Equipo(IdEquipo),
    FOREIGN KEY (IdSoftware) REFERENCES Software(IdSoftware)
);

CREATE TABLE Partes(
    IdParte     INT IDENTITY(1,1) PRIMARY KEY,
    IdEquipo    INT,
    IdTipoParte INT,
    Marca       VARCHAR(50),
    Modelo      VARCHAR(50),
    NroSerie    VARCHAR(100),
    FOREIGN KEY (IdEquipo)    REFERENCES Equipo(IdEquipo),
    FOREIGN KEY (IdTipoParte) REFERENCES Tipo_Parte(IdTipoParte)
);

CREATE TABLE Acceso_Remoto(
    IdAcceso      INT IDENTITY(1,1) PRIMARY KEY,
    IdEquipo      INT,
    SoftwareRemoto VARCHAR(50),
    Codigo        VARCHAR(50),
    Contrasena    VARCHAR(50),
    FOREIGN KEY (IdEquipo) REFERENCES Equipo(IdEquipo)
);

-- PROYECTO 4: Alias para tablas
-- =========================================
SELECT
    D.IdDependencia     AS [ID],
    D.NombreDependencia AS [Nombre de Dependencia]
FROM Dependencia AS D;


-- =========================================
-- PROYECTO 5: Columnas con nombres amigables
-- =========================================
SELECT
    IdEquipo        AS [ID Equipo],
    UsuarioEquipo   AS [Usuario Responsable],
    MarcaModelo     AS [Marca y Modelo],
    Procesador      AS [Procesador],
    RAM_GB          AS [RAM (GB)],
    EstadoEquipo    AS [Estado]
FROM Equipo;


-- =========================================
-- PROYECTO 6: Suma — RAM + 4 GB
-- (RAM_GB es INT, la operación es directa)
-- =========================================
SELECT
    IdEquipo,
    UsuarioEquipo,
    RAM_GB          AS [RAM Actual (GB)],
    RAM_GB + 4      AS [RAM + 4 GB]
FROM Equipo;


-- =========================================
-- PROYECTO 7: Multiplicación — doble de capacidad
-- (CapacidadGB es INT, operación directa)
-- =========================================
SELECT
    IdEquipo,
    UsuarioEquipo,
    CapacidadGB         AS [Capacidad Actual (GB)],
    CapacidadGB * 2     AS [Doble de Capacidad (GB)]
FROM Equipo;


-- =========================================
-- PROYECTO 8: División segura con NULLIF
-- (Precio / RAM_GB, protegido contra cero)
-- =========================================
SELECT
    IdEquipo,
    UsuarioEquipo,
    Precio                          AS [Precio (S/)],
    RAM_GB                          AS [RAM (GB)],
    Precio / NULLIF(RAM_GB, 0)      AS [Precio por GB de RAM]
FROM Equipo;


-- =========================================
-- PROYECTO 9: Precedencia de operadores
-- =========================================
SELECT
    IdEquipo,
    RAM_GB,
    RAM_GB + 4 * 2          AS [Sin paréntesis: RAM + 4*2],
    (RAM_GB + 4) * 2        AS [Con paréntesis: (RAM+4)*2],
    RAM_GB * 2 + CapacidadGB AS [RAM*2 + Capacidad]
FROM Equipo;


-- =========================================
-- PROYECTO 10: Manejo de valores nulos con ISNULL
-- =========================================
SELECT
    IdEquipo,
    UsuarioEquipo,
    ISNULL(Observacion, 'Sin observaciones') AS [Observación],
    ISNULL(GPU, 'Sin GPU dedicada')          AS [GPU],
    ISNULL(MemoriaGPU, 'N/A')               AS [Memoria GPU] 
FROM Equipo;


-- =========================================
-- PROYECTO 11: Concatenar texto simple
-- =========================================
SELECT
    IdEquipo,
    'El equipo N° ' + CAST(IdEquipo AS VARCHAR) +
    ' está asignado a: ' + UsuarioEquipo       AS [Mensaje]
FROM Equipo;


-- =========================================
-- PROYECTO 12: Concatenar marca y procesador
-- =========================================
SELECT
    IdEquipo,
    MarcaModelo + ' — ' + Procesador          AS [Equipo y Procesador]
FROM Equipo;


-- =========================================
-- PROYECTO 13: Concatenar información de hardware
-- =========================================
SELECT
    IdEquipo,
    UsuarioEquipo,
    'RAM: ' + CAST(RAM_GB AS VARCHAR) + ' GB | ' +
    TipoDiscoDuro + ': ' + CAST(CapacidadGB AS VARCHAR) + ' GB' AS [Especificaciones]
FROM Equipo;


-- =========================================
-- PROYECTO 14: Concatenación con valores nulos (usando ISNULL)
-- =========================================
SELECT
    IdEquipo,
    UsuarioEquipo,
    -- Sin protección (puede dar NULL si MemoriaGPU es NULL)
    GPU + ' / ' + MemoriaGPU                                   AS [GPU Info sin protección],
    -- Con protección
    ISNULL(GPU,'Sin GPU') + ' / ' + ISNULL(MemoriaGPU,'N/A')  AS [GPU Info protegida]
FROM Equipo;


-- =========================================
-- PROYECTO 15: COALESCE — primera opción no nula
-- =========================================
SELECT
    IdEquipo,
    UsuarioEquipo,
    COALESCE(GPU, 'Sin GPU dedicada')          AS [GPU],
    COALESCE(MemoriaGPU, 'N/A')               AS [Memoria GPU],
    COALESCE(Observacion, 'Sin observaciones') AS [Observación]
FROM Equipo;


-- =========================================
-- PROYECTO 16: Igualdad — equipos HP
-- =========================================
SELECT
    IdEquipo,
    UsuarioEquipo,
    MarcaModelo,
    EstadoEquipo
FROM Equipo
WHERE MarcaModelo LIKE 'HP%';


-- =========================================
-- PROYECTO 17: Desigualdad — equipos que NO son HP
-- =========================================
SELECT
    IdEquipo,
    UsuarioEquipo,
    MarcaModelo,
    EstadoEquipo
FROM Equipo
WHERE MarcaModelo NOT LIKE 'HP%';


-- =========================================
-- PROYECTO 18: Mayor que — IdEquipo > 10
-- =========================================
SELECT
    IdEquipo,
    UsuarioEquipo,
    MarcaModelo,
    Fecha
FROM Equipo
WHERE IdEquipo > 10;


-- =========================================
-- PROYECTO 19: AND — laptops HP (IdTipoEquipo = 2)
-- =========================================
SELECT
    IdEquipo,
    UsuarioEquipo,
    MarcaModelo,
    EstadoEquipo
FROM Equipo
WHERE MarcaModelo LIKE 'HP%'
  AND IdTipoEquipo = 2;


-- =========================================
-- PROYECTO 20: OR con paréntesis — HP o Dell, tipo Laptop
-- =========================================
SELECT
    IdEquipo,
    UsuarioEquipo,
    MarcaModelo,
    IdTipoEquipo,
    EstadoEquipo
FROM Equipo
WHERE IdTipoEquipo = 2
  AND (MarcaModelo LIKE 'HP%' OR MarcaModelo LIKE 'Dell%');


-- =========================================
-- PROYECTO 21: LIKE con % — usuarios que empiezan con 'U'
-- =========================================
SELECT
    IdEquipo,
    UsuarioEquipo,
    MarcaModelo
FROM Equipo
WHERE UsuarioEquipo LIKE 'U%';


-- =========================================
-- PROYECTO 22: LIKE con _ — 2do carácter es 'a'
-- =========================================
SELECT
    IdEquipo,
    UsuarioEquipo
FROM Equipo
WHERE UsuarioEquipo LIKE '_a%';


-- =========================================
-- PROYECTO 23: BETWEEN — equipos con ID entre 5 y 15
-- =========================================
SELECT
    IdEquipo,
    UsuarioEquipo,
    MarcaModelo,
    Fecha
FROM Equipo
WHERE IdEquipo BETWEEN 5 AND 15;


-- =========================================
-- PROYECTO 24: IN con valores explícitos — series específicas
-- =========================================
SELECT
    IdEquipo,
    NroSerieCPU,
    MarcaModelo,
    UsuarioEquipo
FROM Equipo
WHERE NroSerieCPU IN ('CPU001','CPU010','CPU020','CPU030','CPU040','CPU050');


-- =========================================
-- PROYECTO 25: IN con subconsulta — equipos en dependencias existentes
-- =========================================
SELECT
    IdEquipo,
    UsuarioEquipo,
    IdDependencia
FROM Equipo
WHERE IdDependencia IN (
    SELECT IdDependencia FROM Dependencia
);


-- =========================================
-- PROYECTO 26: Extraer año de la fecha
-- =========================================
SELECT
    IdEquipo,
    UsuarioEquipo,
    Fecha,
    YEAR(Fecha) AS [Año de Registro]
FROM Equipo;


-- =========================================
-- PROYECTO 27: Extraer mes y día
-- =========================================
SELECT
    IdEquipo,
    UsuarioEquipo,
    Fecha,
    MONTH(Fecha) AS [Mes],
    DAY(Fecha)   AS [Día]
FROM Equipo;


-- =========================================
-- PROYECTO 28: Agregar días a la fecha
-- =========================================
SELECT
    IdEquipo,
    UsuarioEquipo,
    Fecha                            AS [Fecha de Registro],
    DATEADD(DAY,   365, Fecha)       AS [+ 1 Año],
    DATEADD(MONTH,   6, Fecha)       AS [+ 6 Meses]
FROM Equipo;


-- =========================================
-- PROYECTO 29: Diferencia entre fechas
-- =========================================
SELECT
    IdEquipo,
    UsuarioEquipo,
    Fecha                                    AS [Fecha Registro],
    CAST(GETDATE() AS DATE)                  AS [Hoy],
    DATEDIFF(DAY,   Fecha, GETDATE())        AS [Días Transcurridos],
    DATEDIFF(MONTH, Fecha, GETDATE())        AS [Meses Transcurridos],
    DATEDIFF(YEAR,  Fecha, GETDATE())        AS [Años Transcurridos]
FROM Equipo;


-- =========================================
-- PROYECTO 30: Convertir formato de fechas
-- =========================================
SELECT
    IdEquipo,
    UsuarioEquipo,
    Fecha                                       AS [Original],
    CONVERT(VARCHAR, Fecha, 103)                AS [DD/MM/YYYY],
    CONVERT(VARCHAR, Fecha, 112)                AS [YYYYMMDD],
    FORMAT(Fecha, 'dd-MM-yyyy')                 AS [DD-MM-YYYY],
    FORMAT(Fecha, 'MMMM dd, yyyy', 'es-PE')    AS [Formato Largo Español]
FROM Equipo;
